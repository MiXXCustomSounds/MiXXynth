# M7: the factory preset bank, embedded in the binary.
#
# The XMLs live in `presets/` in the repository as text - no binary assets, no licensing
# question - and are compiled in rather than copied to disk at runtime, because a VST3
# installed into a DAW has no repository to copy from. The glob is CONFIGURE_DEPENDS so adding
# a preset re-runs CMake instead of silently leaving it out of the bank.
#
# Generated by tools/make-factory-presets.mjs; see docs/PLAN-M7.md.
file(GLOB MIXXYNTH_PRESETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/presets/*.xml")

list(LENGTH MIXXYNTH_PRESETS MIXXYNTH_NUM_PRESETS)

if(MIXXYNTH_NUM_PRESETS LESS 1)
    message(FATAL_ERROR "No factory presets found in presets/. The bank is part of the build.")
endif()

juce_add_binary_data(MiXXynthPresetData
    NAMESPACE MiXXPresets
    HEADER_NAME MiXXPresets.h
    SOURCES ${MIXXYNTH_PRESETS}
)

# The smoke test holds a MiXXVoiceManager on the stack, and that is ~52 KB. The default
# 1 MB stack left so little headroom that adding 13 KB to the manager overflowed it - and
# the failure mode was a crash BEFORE main with no output whatsoever, which is close to
# undebuggable. An explicit reserve makes the budget deliberate instead of accidental.
if (MSVC)
    target_link_options(MiXXynthSmoke PRIVATE /STACK:8388608)
endif()
target_compile_definitions(MiXXynthSmoke PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_STANDALONE_APPLICATION=0
)

# M8: the FX measurement harness. A separate target because it is a HOST rather than a plugin:
# it loads a VST3 from an absolute path, renders through it offline, and measures the result.
#
# JUCE_PLUGINHOST_VST3 is what enables the hosting half of juce_audio_processors, and it is off
# everywhere else in this project deliberately - MiXXynth hosts nothing, and enabling it there
# would link a plugin scanner into a plugin.
#
# This is a console program rather than a gate: it takes arguments, it measures things outside
# this repository, and a missing plugin on a given machine is not a build failure.
add_executable(MiXXynthFxMeasure
    tests/FxMeasure.cpp
    Source/FxMeasure/TestSignals.cpp
    Source/FxMeasure/Analysis.cpp
    Source/FxMeasure/PluginHost.cpp
)
target_link_libraries(MiXXynthFxMeasure PRIVATE
    juce::juce_audio_utils
    juce::juce_dsp
    juce::juce_gui_basics
    PUBLIC juce::juce_recommended_config_flags
           juce::juce_recommended_warning_flags
)
target_compile_features(MiXXynthFxMeasure PRIVATE cxx_std_20)
target_compile_definitions(MiXXynthFxMeasure PRIVATE
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_STANDALONE_APPLICATION=0
    JUCE_PLUGINHOST_VST3=1
)

if (MSVC)
    # A sweep measurement holds several transform-sized buffers at once - a three-second sweep
    # at 48 kHz deconvolves through a 262144-point transform - and the self-test holds more on
    # top of that. Same reasoning as MiXXynthSmoke: an explicit budget rather than an accidental
    # one, because the failure mode without it is a crash before any output at all.
    target_link_options(MiXXynthFxMeasure PRIVATE /STACK:16777216)
endif()
